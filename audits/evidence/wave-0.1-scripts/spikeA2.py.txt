import os, sys, json, time, random, statistics
from datetime import datetime, timedelta, timezone
from zoneinfo import ZoneInfo
from urllib.parse import urlparse
import requests, pyarrow.parquet as pq

UA = os.environ.get("SEC_USER_AGENT", "")
if not UA:
    sys.exit("SEC_USER_AGENT is not set; refusing to send any request (IG-08)")
_last = [0.0]
def sec_get(url):
    host = urlparse(url).hostname or ""
    assert host == "sec.gov" or host.endswith(".sec.gov"), host   # contact string goes to sec.gov only
    wait = 0.25 - (time.time() - _last[0])
    if wait > 0: time.sleep(wait)                                  # at most 4 requests a second
    for attempt in range(3):
        r = requests.get(url, headers={"User-Agent": UA, "Accept-Encoding": "gzip"}, timeout=30)
        _last[0] = time.time()
        if r.status_code == 200: return r.json()
        if r.status_code in (429, 500, 502, 503): time.sleep(2 ** attempt); continue
        return None
    return None

tick = sec_get("https://www.sec.gov/files/company_tickers.json")
cik_of = {v["ticker"].upper(): int(v["cik_str"]) for v in tick.values()}

rows = pq.ParquetFile("hf/part-0.parquet").read(columns=["symbol","company_name","year","quarter","date"]).to_pylist()
by_year = {}
for r in rows:
    y = int(r["date"][:4]); r["cy"] = y
    if 2015 <= y <= 2024 and r["symbol"].upper() in cik_of: by_year.setdefault(y, []).append(r)
rng = random.Random(2027)
sample = []
for y in range(2015, 2025):
    sample += rng.sample(sorted(by_year[y], key=lambda r:(r["symbol"],r["date"])), 5)
print("sampled", len(sample), "from resolvable tickers; per-year pool sizes", {y: len(v) for y, v in sorted(by_year.items())})

subs_cache = {}
def filings_for(cik):
    if cik in subs_cache: return subs_cache[cik]
    j = sec_get(f"https://data.sec.gov/submissions/CIK{cik:010d}.json")
    out = []
    if j:
        def add(block):
            for i in range(len(block["form"])):
                out.append((block["form"][i], block["items"][i], block["filingDate"][i], block["acceptanceDateTime"][i]))
        add(j["filings"]["recent"])
        for f in j["filings"].get("files", []):
            if f["filingTo"] >= "2014-12-01":
                k = sec_get("https://data.sec.gov/submissions/" + f["name"])
                if k: add(k)
    subs_cache[cik] = out
    return out

res = []
for r in sample:
    cik = cik_of[r["symbol"].upper()]
    call_local = datetime.strptime(r["date"], "%Y-%m-%d %H:%M:%S")
    cands = []
    for form, items, fdate, acc in filings_for(cik):
        if form.startswith("8-K") and "2.02" in (items or "").split(","):
            fd = datetime.strptime(fdate, "%Y-%m-%d")
            if abs((fd.date() - call_local.date()).days) <= 3:
                acc_utc = datetime.strptime(acc[:19], "%Y-%m-%dT%H:%M:%S").replace(tzinfo=timezone.utc)
                cands.append(acc_utc)
    rec = {"symbol": r["symbol"], "call_raw": r["date"], "cy": r["cy"], "matched": bool(cands)}
    if cands:
        acc = min(cands, key=lambda a: abs((a - call_local.replace(tzinfo=timezone.utc)).total_seconds()))
        call_et = call_local.replace(tzinfo=ZoneInfo("America/New_York")).astimezone(timezone.utc)
        call_utc = call_local.replace(tzinfo=timezone.utc)
        rec["acc_utc"] = acc.isoformat()
        rec["gap_min_if_eastern"] = (acc - call_et).total_seconds() / 60    # positive: filing after call
        rec["gap_min_if_utc"] = (acc - call_utc).total_seconds() / 60
    res.append(rec)
json.dump(res, open("spikeA_results.json", "w"), indent=1)

m = [x for x in res if x["matched"]]
print("matched", len(m), "of", len(res))
print("unmatched:", [(x["symbol"], x["call_raw"][:10]) for x in res if not x["matched"]])
for tag in ("gap_min_if_eastern", "gap_min_if_utc"):
    g = [x[tag] for x in m]
    first = sum(1 for v in g if v > 0)
    q = statistics.quantiles(g, n=4) if len(g) >= 4 else []
    print(tag, "call first in", first, "of", len(g), "| median", round(statistics.median(g)), "min", round(min(g)), "max", round(max(g)), "quartiles", [round(v) for v in q])
