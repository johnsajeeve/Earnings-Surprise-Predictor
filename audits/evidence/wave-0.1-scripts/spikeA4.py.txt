    w=0.25-(time.time()-_l[0])
    if w>0: time.sleep(w)

tick=sec_get("https://www.sec.gov/files/company_tickers.json")
cik_of={v["ticker"].upper():int(v["cik_str"]) for v in tick.values()}
rows=pq.ParquetFile("hf/part-0.parquet").read(columns=["symbol","date"]).to_pylist()

sel=[r for r in rows if "2015"<=r["date"][:4]<="2024" and r["date"][11:16] in ("16:30","17:00") and r["symbol"].upper() in cik_of]

cache={}
def filings(cik):
    if cik in cache: return cache[cik]
    j=sec_get(f"https://data.sec.gov/submissions/CIK{cik:010d}.json"); out=[]
    if j:
        def add(b):
            for i in range(len(b["form"])): out.append((b["form"][i],b["items"][i],b["filingDate"][i],b["acceptanceDateTime"][i]))
        add(j["filings"]["recent"])
        for f in j["filings"].get("files",[]):
            if f["filingTo"]>="2014-12-01":
                k=sec_get("https://data.sec.gov/submissions/"+f["name"])
                if k: add(k)
    cache[cik]=out; return out
ET=ZoneInfo("America/New_York"); ev=[]
for n,r in enumerate(sorted(sel,key=lambda r:(r["symbol"],r["date"]))):
    c=datetime.strptime(r["date"],"%Y-%m-%d %H:%M:%S"); best=None
    for form,items,fd,acc in filings(cik_of[r["symbol"].upper()]):
        if form.startswith("8-K") and "2.02" in (items or "").split(","):
            dd=abs((datetime.strptime(fd,"%Y-%m-%d").date()-c.date()).days)
            a=datetime.strptime(acc[:19],"%Y-%m-%dT%H:%M:%S").replace(tzinfo=timezone.utc)
            if dd<=3 and (best is None or (dd,a)<best[:2]): best=(dd,a)
    if best:
        aet=best[1].astimezone(ET)
        if (aet.hour==16 and aet.minute<=30) and aet.date()==c.date(): ev.append((r["symbol"],c,best[1]))
    if n%500==0: print("progress",n,"of",len(sel),"anchor events so far",len(ev),flush=True)
print("anchor events (filing 16:00-16:30 ET on the call date):",len(ev))
H={"H1 Eastern with DST":lambda c:c.replace(tzinfo=ET).astimezone(timezone.utc),
   "H2 Eastern standard all year (UTC-5)":lambda c:(c+timedelta(hours=5)).replace(tzinfo=timezone.utc)}
res={}
for name,f in H.items():
    s=[];w=[]
    for sym,c,a in ev:
        g=(a-f(c)).total_seconds()/60
        if 4<=c.month<=10: s.append(g)
        elif c.month in (12,1,2): w.append(g)
    res[name]={"n_summer":len(s),"n_winter":len(w),"median_summer":round(statistics.median(s)),"median_winter":round(statistics.median(w)),"abs_diff":round(abs(statistics.median(s)-statistics.median(w)))}
    print(name,res[name])
json.dump(res,open("spikeA4_results.json","w"),indent=1)
