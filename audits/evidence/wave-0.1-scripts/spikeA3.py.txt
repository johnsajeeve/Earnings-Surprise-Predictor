import os, sys, json, time, random, statistics, collections
from datetime import datetime, timezone, timedelta
from zoneinfo import ZoneInfo
from urllib.parse import urlparse
import requests, pyarrow.parquet as pq
UA=os.environ.get("SEC_USER_AGENT","")
if not UA: sys.exit("SEC_USER_AGENT not set")
_l=[0.0]
def sec_get(url):
    h=urlparse(url).hostname or ""; assert h=="sec.gov" or h.endswith(".sec.gov"), h
    w=0.25-(time.time()-_l[0])
    if w>0: time.sleep(w)
    for a in range(3):
        r=requests.get(url,headers={"User-Agent":UA},timeout=30); _l[0]=time.time()
        if r.status_code==200: return r.json()
        if r.status_code in (429,500,502,503): time.sleep(2**a); continue
        return None
tick=sec_get("https://www.sec.gov/files/company_tickers.json")
cik_of={v["ticker"].upper():int(v["cik_str"]) for v in tick.values()}
rows=pq.ParquetFile("hf/part-0.parquet").read(columns=["symbol","date"]).to_pylist()
# Test 1: whole set
tod=collections.Counter(r["date"][11:16] for r in rows if r["date"][11:19]!="00:00:00")
tot=sum(tod.values())
def share(lo,hi): return sum(n for t,n in tod.items() if lo<=t<=hi)/tot
print("Test 1: non-midnight rows",tot,"| share 08:00-17:30 as stamped:",round(100*share("08:00","17:30"),1),"%")
print("  most common stamped times:",tod.most_common(8))
# if the stamp were UTC, the same instants in US Eastern would be 4-5 h earlier; if London, 5 h later than Eastern
def shifted(h):
    c=0
    for t,n in tod.items():
        hh=int(t[:2])*60+int(t[3:]); hh=(hh+h*60)%1440
        if 8*60<=hh<=17*60+30: c+=n
    return c/tot
print("  business-hours share if the stamp is UTC (read in ET, -5h):",round(100*shifted(-5),1),"%;  if the stamp is London (read in ET, -5h): same shift as UTC in winter")
by={}
for r in rows:
    y=int(r["date"][:4])
    if 2015<=y<=2024 and r["symbol"].upper() in cik_of and r["date"][11:19]!="00:00:00": by.setdefault(y,[]).append(r)
rng=random.Random(2028); sample=[]
for y in range(2015,2025): sample+=rng.sample(sorted(by[y],key=lambda r:(r["symbol"],r["date"])),40)
cache={}
def filings(cik):
    if cik in cache: return cache[cik]
    j=sec_get(f"https://data.sec.gov/submissions/CIK{cik:010d}.json"); out=[]
    if j:
        def add(b):
            for i in range(len(b["form"])): out.append((b["form"][i],b["items"][i],b["filingDate"][i],b["acceptanceDateTime"][i]))
        add(j["filings"]["recent"])
        for f in j["filings"].get("files",[]):
            if f["filingTo"]>="2014-12-01":
                k=sec_get("https://data.sec.gov/submissions/"+f["name"])
                if k: add(k)
    cache[cik]=out; return out
ev=[]
for r in sample:
    c=datetime.strptime(r["date"],"%Y-%m-%d %H:%M:%S"); best=None
    for form,items,fd,acc in filings(cik_of[r["symbol"].upper()]):
        if form.startswith("8-K") and "2.02" in (items or "").split(","):
            dd=abs((datetime.strptime(fd,"%Y-%m-%d").date()-c.date()).days)
            a=datetime.strptime(acc[:19],"%Y-%m-%dT%H:%M:%S").replace(tzinfo=timezone.utc)
            if dd<=3 and (best is None or (dd,a)<best[:2]): best=(dd,a)
    if best: ev.append((r["symbol"],c,best[1]))
print("matched",len(ev),"of",len(sample))
H={"H1 Eastern with DST":lambda c:c.replace(tzinfo=ZoneInfo("America/New_York")).astimezone(timezone.utc),
   "H2 Eastern standard all year (UTC-5)":lambda c:(c+timedelta(hours=5)).replace(tzinfo=timezone.utc),
   "H3 London with DST":lambda c:c.replace(tzinfo=ZoneInfo("Europe/London")).astimezone(timezone.utc),
   "H4 UTC":lambda c:c.replace(tzinfo=timezone.utc)}
res={}
for name,f in H.items():
    summer=[];winter=[];allg=[]
    for s,c,a in ev:
        g=(a-f(c)).total_seconds()/60
        if abs(g)>1440: continue
        allg.append(g)
        if 4<=c.month<=10: summer.append(g)
        elif c.month in (12,1,2): winter.append(g)
    ms,mw=statistics.median(summer),statistics.median(winter)
    res[name]={"n_within_24h":len(allg),"n_summer":len(summer),"n_winter":len(winter),"median_summer":round(ms),"median_winter":round(mw),"abs_diff":round(abs(ms-mw)),"call_first_share":round(100*sum(g>0 for g in allg)/len(allg),1)}
    print(name,res[name])
json.dump(res,open("spikeA3_results.json","w"),indent=1)
